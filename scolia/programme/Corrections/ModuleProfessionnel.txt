Option Explicit
Private mMajuscule As Boolean
Public Sub MajusculeChamp(ByVal champ As Object)
    Dim pos As Long, valeur As String
    If mMajuscule Then Exit Sub
    On Error GoTo Fin
    valeur = CStr(champ.Value)
    If valeur = UCase$(valeur) Then Exit Sub
    mMajuscule = True
    pos = champ.SelStart
    champ.Value = UCase$(valeur)
    champ.SelStart = pos
Fin:
    mMajuscule = False
End Sub
Public Function ValeurFiltre(ByVal f As Object, ByVal nom1 As String, ByVal nom2 As String) As String
    On Error Resume Next
    ValeurFiltre = CStr(f.Controls(nom1).Value)
    If Err.Number <> 0 Then
        Err.Clear
        ValeurFiltre = CStr(f.Controls(nom2).Value)
    End If
    On Error GoTo 0
End Function

Private Sub StyliserControles(ByVal parent As Object)
    Dim c As Object
    For Each c In parent.Controls
        Select Case TypeName(c)
            Case "TextBox"
                c.Font.Name = "Segoe UI": c.Font.Size = 10
                c.ForeColor = RGB(30, 41, 59)
                c.SpecialEffect = fmSpecialEffectFlat
                c.BorderStyle = fmBorderStyleSingle
                c.BorderColor = RGB(211, 213, 219)
                If c.Locked Then
                    c.BackColor = RGB(232, 237, 243): c.TabStop = False
                Else
                    c.BackColor = vbWhite
                End If
            Case "ComboBox", "ListBox"
                c.Font.Name = "Segoe UI": c.Font.Size = 10
                c.BackColor = vbWhite: c.ForeColor = RGB(30, 41, 59)
                c.SpecialEffect = fmSpecialEffectFlat
                c.BorderStyle = fmBorderStyleSingle
                c.BorderColor = RGB(211, 213, 219)
            Case "CommandButton"
                c.Font.Name = "Segoe UI": c.Font.Size = 10
                c.BackColor = RGB(0, 112, 235): c.ForeColor = vbWhite
                c.Font.Bold = True
                EtiqueterBoutonPro c
            ColorerBoutonPro c
            Case "Label"
                If c.Name <> "LabelTitrePro" And c.Name <> "LabelInstitutionPro" Then c.Font.Size = 9
                c.BackStyle = fmBackStyleTransparent
                c.Font.Name = "Segoe UI"
                c.ForeColor = RGB(30, 41, 59)
            Case "Frame"
                c.BackColor = RGB(246, 246, 248)
                StyliserControles c
        End Select
    Next c
End Sub

Public Function ContactValide(ByVal valeur As String, ByVal obligatoire As Boolean) As Boolean
    Dim i As Long
    valeur = Trim$(valeur)
    If valeur = "" Then ContactValide = Not obligatoire: Exit Function
    If Len(valeur) <> 10 Then Exit Function
    For i = 1 To 10
        If Mid$(valeur, i, 1) < "0" Or Mid$(valeur, i, 1) > "9" Then Exit Function
    Next i
    ContactValide = True
End Function

Public Sub HabillerRapportPDF(ByVal w As Worksheet, ByVal entete As Long, ByVal Total As Long, ByVal colonnes As Long, Optional ByVal ligneTotal As Boolean = True)
    Dim i As Long, j As Long, derniereDonnee As Long, zone As Range, label As String
    Set zone = w.Range(w.Cells(1, 1), w.Cells(Total, colonnes))
    With zone
        .Font.Name = "Segoe UI": .Font.Size = 10
        .Font.Color = RGB(31, 41, 55): .VerticalAlignment = xlCenter
        .HorizontalAlignment = xlLeft: .WrapText = True
    End With
    With w.Range(w.Cells(1, 1), w.Cells(1, colonnes))
        .Interior.Color = RGB(38, 91, 155): .Font.Color = vbWhite
        .Font.Size = 13: .Font.Bold = True: .RowHeight = 36
        .HorizontalAlignment = xlCenter
    End With
    With w.Range(w.Cells(2, 1), w.Cells(2, colonnes))
        .Font.Size = 12: .Font.Bold = True: .Font.Color = RGB(31, 41, 55)
        .RowHeight = 28: .HorizontalAlignment = xlLeft
    End With
    w.Rows(3).RowHeight = 23: w.Rows(3).Font.Size = 9
    w.Rows(3).Font.Color = RGB(91, 105, 122)
    With w.Range(w.Cells(entete, 1), w.Cells(entete, colonnes))
        .Interior.Color = RGB(38, 91, 155): .Font.Color = vbWhite
        .Font.Bold = True: .WrapText = True: .RowHeight = 36
    End With
    derniereDonnee = Total
    If ligneTotal Then derniereDonnee = Total - 1
    If derniereDonnee > entete Then w.Range(w.Cells(entete + 1, 1), w.Cells(derniereDonnee, colonnes)).Rows.AutoFit
    For i = entete + 1 To derniereDonnee
        With w.Range(w.Cells(i, 1), w.Cells(i, colonnes))
            If (i - entete) Mod 2 = 0 Then .Interior.Color = RGB(249, 250, 252)
            .Borders(xlEdgeBottom).LineStyle = xlContinuous
            .Borders(xlEdgeBottom).Color = RGB(224, 228, 234)
            .Borders(xlEdgeBottom).Weight = xlHairline
        End With
        If w.Rows(i).RowHeight < 26 Then w.Rows(i).RowHeight = 26
    Next i
    For j = 1 To colonnes
        label = UCase$(CStr(w.Cells(entete, j).Value))
        If InStr(1, label, " FC", vbTextCompare) > 0 Then
            w.Range(w.Cells(entete + 1, j), w.Cells(Total, j)).HorizontalAlignment = xlRight
            w.Range(w.Cells(entete + 1, j), w.Cells(Total, j)).NumberFormat = "#,##0"
        End If
    Next j
    If ligneTotal Then
        With w.Range(w.Cells(Total, 1), w.Cells(Total, colonnes))
            .Interior.Color = RGB(229, 238, 248): .Font.Bold = True
            .Font.Color = RGB(31, 41, 55): .RowHeight = 30
            .Borders(xlEdgeTop).LineStyle = xlContinuous
            .Borders(xlEdgeTop).Color = RGB(38, 91, 155)
        End With
    End If
    With w.PageSetup
        .PaperSize = xlPaperA4: .Orientation = xlLandscape
        .Zoom = False: .FitToPagesWide = 1: .FitToPagesTall = False
        .CenterHorizontally = True
        .LeftMargin = Application.CentimetersToPoints(1.2): .RightMargin = Application.CentimetersToPoints(1.2)
        .TopMargin = Application.CentimetersToPoints(1.2): .BottomMargin = Application.CentimetersToPoints(1.2)
        .HeaderMargin = Application.CentimetersToPoints(0.5): .FooterMargin = Application.CentimetersToPoints(0.6)
        .PrintGridlines = False: .PrintHeadings = False
        .LeftFooter = "SCOLIA | " & w.Name: .CenterFooter = "": .RightFooter = "Page &P / &N"
    End With
End Sub

Private Sub ReorganiserFormulaire(ByVal f As Object)
    Dim ordre As Variant, noms As Variant, nom As Variant, c As Object
    Dim utilises As Object, i As Long, y As Single, colonnes As Long
    Dim largeur As Single, hauteur As Single
    If f.Name = "Apprenant" Then
        DisposerInscription f
        Exit Sub
    End If
    largeur = Application.Min(820, Application.Width - 48)
    If largeur < 240 Then largeur = 240
    f.Width = largeur + 24
    f.Controls("LabelTitrePro").Width = largeur - 24
    f.Controls("LabelInstitutionPro").Width = largeur - 24
    Set utilises = CreateObject("Scripting.Dictionary")
    Select Case f.Name
        Case "Apprenant": ordre = PlanApprenant()
        Case "Paiement": ordre = PlanPaiement()
        Case "TableauDeBord": ordre = PlanTableauDeBord()
        Case "GestionApprenant": ordre = PlanGestionApprenant()
        Case Else: Exit Sub
    End Select
    ' Les anciens libelles sont remplaces par des libelles attaches a chaque champ.
    For Each c In f.Controls
        If TypeName(c) = "Label" Then
            If c.Name <> "LabelTitrePro" And c.Name <> "LabelInstitutionPro" Then c.Visible = False
        End If
    Next c
    y = 54: i = 0: colonnes = 4
    If largeur < 700 Then colonnes = 3
    If largeur < 480 Then colonnes = 2
    If (f.Name = "GestionApprenant" Or f.Name = "Paiement") And largeur >= 700 Then colonnes = 4
    For Each nom In ordre
        If Left$(CStr(nom), 1) = "@" Then
            If i > 0 Then y = y + 40: i = 0
            AjouterSectionPro f, Mid$(CStr(nom), 2), largeur, y
        Else
            Set c = ControleDisposition(f, CStr(nom))
            If Not c Is Nothing Then
                If c.Visible Then
                    PlacerChampPro f, c, largeur, y, i, colonnes
                    utilises(c.Name) = True
                End If
            End If
        End If
    Next nom
    ' Ne pas exposer de controles anciens ou inconnus hors de la liste du formulaire.
    For Each c In f.Controls
        If Not utilises.Exists(c.Name) Then
            Select Case TypeName(c)
                Case "TextBox", "ComboBox", "ListBox", "Frame", "MultiPage"
                    c.Visible = False
            End Select
        End If
    Next c
    If i > 0 Then y = y + 40
    y = y + 6: i = 0
    For Each c In f.Controls
        If TypeName(c) = "CommandButton" And c.Visible Then
            c.Left = 12 + (i Mod colonnes) * (largeur / colonnes)
            c.Top = y + (i \ colonnes) * 30
            c.Width = largeur / colonnes - 12: c.Height = 26
            c.Font.Size = 9: c.WordWrap = True
            EtiqueterBoutonPro c
            ColorerBoutonPro c
            i = i + 1
        End If
    Next c
    y = y + ((i + colonnes - 1) \ colonnes) * 30 + 12
    hauteur = Application.Height - 48
    If hauteur > 760 Then hauteur = 760
    f.Height = Application.Min(y + 36, hauteur)
    AdapterHauteurSansDefilement f, y
    f.ScrollTop = 0
End Sub
Private Function ControleDisposition(ByVal f As Object, ByVal nom As String) As Object
    On Error Resume Next
    Set ControleDisposition = f.Controls(nom)
    On Error GoTo 0
End Function
Private Sub PlacerChampPro(ByVal f As Object, ByVal c As Object, ByVal largeur As Single, ByRef y As Single, ByRef i As Long, ByVal colonnes As Long)
    Dim libelle As Object, x As Single, titre As String, grand As Boolean
    grand = TypeName(c) = "ListBox" Or TypeName(c) = "Frame" Or TypeName(c) = "MultiPage" Or c.Name = "TextAffectation"
    If grand And i > 0 Then y = y + 40: i = 0
    x = 12 + i * (largeur / colonnes)
    c.Left = x: c.Top = y + 16
    c.Width = largeur / colonnes - 12: c.Height = 22
    If grand Then
        c.Left = 12: c.Width = largeur - 24: c.Height = 64
        If TypeName(c) = "Frame" Or TypeName(c) = "MultiPage" Then c.Height = 160
    End If
    Set libelle = ControleDisposition(f, "ProLibelle_" & c.Name)
    If libelle Is Nothing Then Set libelle = f.Controls.Add("Forms.Label.1", "ProLibelle_" & c.Name, True)
    titre = LibelleChampPro(c.Name)
    If f.Name = "GestionApprenant" Then
        If c.Name = "ListStatutApprenant" Then titre = "Filtrer par statut"
        If c.Name = "ListStatut" Then titre = "Nouveau statut de l'apprenant"
    End If
    If f.Name = "Apprenant" And c.Name = "ListPromotion" Then titre = "Promotion"
    If f.Name = "Apprenant" And c.Name = "ListStatut" Then titre = "Statut de l'apprenant"
    With libelle
        .Caption = titre: .Left = c.Left: .Top = y
        .Width = c.Width: .Height = 16
        .Font.Name = "Segoe UI": .Font.Size = 9: .Font.Bold = True
        .ForeColor = RGB(71, 85, 105): .BackStyle = fmBackStyleTransparent
        .Visible = True
    End With
    If c.Name = "TextAdresse" Then c.MultiLine = False
    If c.Name = "TextAffectation" Then c.Height = 48: c.MultiLine = True
    If grand Then
        y = y + c.Height + 23: i = 0
    Else
        i = i + 1
        If i = colonnes Then y = y + 40: i = 0
    End If
End Sub

        c.BackColor = RGB(74, 190, 112): c.ForeColor = vbBlack
    End If
    If InStr(action, "fermer") > 0 Or InStr(action, "femer") > 0 Or InStr(action, "ferme") > 0 Then
        c.BackColor = RGB(192, 48, 48): c.ForeColor = vbWhite
    End If
    If c.Name = "CommandButton2" Then
        c.BackColor = RGB(192, 48, 48): c.ForeColor = vbWhite
    End If
End Sub

Private Sub AdapterHauteurSansDefilement(ByVal f As Object, ByVal bas As Single)
    Dim c As Object, n As Long, i As Long, nbColonnes As Long, lignes As Long
    Dim debutBoutons As Single, sommet As Single, basContenu As Single, hauteurTotale As Single
    debutBoutons = bas
    For Each c In f.Controls
        If c.Visible And TypeName(c) = "CommandButton" Then
            n = n + 1
            If c.Top < debutBoutons Then debutBoutons = c.Top
        End If
    Next c
    ' Aucune compression verticale des champs ou de leurs textes.
    For Each c In f.Controls
        If c.Visible And TypeName(c) <> "CommandButton" Then
            If c.Top + c.Height > basContenu Then basContenu = c.Top + c.Height
        End If
    Next c
    nbColonnes = 3
    If f.InsideWidth < 480 Then nbColonnes = 2
    lignes = (n + nbColonnes - 1) \ nbColonnes
    sommet = basContenu + 12
    For Each c In f.Controls
        If c.Visible And TypeName(c) = "CommandButton" Then
            c.Top = sommet + (i \ nbColonnes) * 30
            c.Left = 12 + (i Mod nbColonnes) * ((f.InsideWidth - 24) / nbColonnes)
            c.Width = (f.InsideWidth - 24) / nbColonnes - 8
            c.Height = 26: c.Font.Size = 9
            c.WordWrap = True: c.AutoSize = False
            c.ControlTipText = c.Caption
            i = i + 1
        End If
    Next c
    hauteurTotale = sommet + lignes * 30 + 12
    f.Height = Application.Min(hauteurTotale + (f.Height - f.InsideHeight), Application.Height - 48)
    f.ScrollHeight = hauteurTotale
    If hauteurTotale > f.InsideHeight Then
        f.ScrollBars = fmScrollBarsVertical
    Else
        f.ScrollBars = fmScrollBarsNone
    End If
    f.StartUpPosition = 0
    f.Left = Application.Left + (Application.Width - f.Width) / 2
    f.Top = Application.Top + (Application.Height - f.Height) / 2
End Sub

Public Sub ChargerPromotionsValides(ByVal liste As Object, Optional ByVal toutes As Boolean = False)
    Dim ws As Worksheet, i As Long, code As String, d As Object
    Set ws = ThisWorkbook.Worksheets("PROMOTIONS")
    Set d = CreateObject("Scripting.Dictionary")
    liste.Clear
    If toutes Then liste.AddItem "Toutes les promotions"
    For i = 5 To ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
        code = UCase$(Trim$(CStr(ws.Cells(i, 1).Value)))
        If code <> "" And Not d.Exists(code) Then
            d.Add code, True: liste.AddItem code
        End If
    Next i
End Sub

Private Sub FinaliserChampsPro(ByVal f As Object)
    Dim c As Object, cle As String
    For Each c In f.Controls
        If c.Visible Then
            cle = LCase$(c.Name)
            Select Case TypeName(c)
                Case "TextBox"
                    c.SpecialEffect = fmSpecialEffectFlat
                    c.BorderStyle = fmBorderStyleSingle
                    c.BorderColor = RGB(210, 213, 220)
                    c.ForeColor = RGB(35, 37, 42)
                    c.Font.Name = "Segoe UI"
                    c.TextAlign = fmTextAlignLeft
                    If c.Locked Then
                        c.BackColor = RGB(239, 241, 245)
                        c.Font.Bold = True
                    Else
                        c.BackColor = vbWhite
                        c.Font.Bold = False
                    End If
                    If InStr(cle, "montant") > 0 Or InStr(cle, "revenu") > 0 Or InStr(cle, "total") > 0 Or InStr(cle, "reste") > 0 Or InStr(cle, "dettes") > 0 Or InStr(cle, "taux") > 0 Then c.TextAlign = fmTextAlignRight
                    If InStr(cle, "date") > 0 Or cle = "textmatricule" Then c.TextAlign = fmTextAlignCenter
                    If cle = "textrecherche" Then
                        c.BorderColor = RGB(0, 112, 235)
                        c.ControlTipText = "Saisissez le matricule ou le nom pour rechercher un apprenant."
                    ElseIf Len(c.ControlTipText) = 0 Then
                        c.ControlTipText = LibelleChampPro(c.Name)
                    End If
                Case "ComboBox"
                    c.SpecialEffect = fmSpecialEffectFlat
                    c.BackColor = vbWhite: c.BorderColor = RGB(210, 213, 220)
                    c.Font.Name = "Segoe UI"
                    c.ControlTipText = LibelleChampPro(c.Name)
                Case "ListBox"
                    c.IntegralHeight = False
                    c.SpecialEffect = fmSpecialEffectFlat
                    c.BorderStyle = fmBorderStyleSingle
                    c.BorderColor = RGB(210, 213, 220)
                    c.BackColor = vbWhite: c.ForeColor = RGB(35, 37, 42)
                    c.Font.Name = "Segoe UI"
                    c.ControlTipText = LibelleChampPro(c.Name)
            End Select
        End If
    Next c
End Sub

Private Sub DisposerInscription(ByVal f As Object)
    Dim champs As Variant, nom As Variant, c As Object, l As Object
    Dim largeur As Single, col As Single, x As Single, y As Single
    Dim i As Long, n As Long, bas As Single
    largeur = Application.Min(720, Application.Width - 48)
    f.Width = largeur + 18
    largeur = f.InsideWidth - 24: col = (largeur - 24) / 4
    For Each c In f.Controls
        If TypeName(c) = "Label" Then c.Visible = False
    Next c
    With f.Controls("LabelTitrePro")
        .Visible = True: .Left = 12: .Top = 10: .Width = largeur: .Height = 24
        .Caption = "Inscription d'un apprenant": .Font.Size = 15
    End With
    With f.Controls("LabelInstitutionPro")
        .Visible = True: .Left = 12: .Top = 35: .Width = largeur: .Height = 17
    End With
    champs = Array("TextMatricule", "TextNomApprenant", "TextPostNom", "TextPrenom", "ListSexe", "TextDateNaissance", "TextLieuNaissance", "TextAdresse", "TextTuteur", "TextParent1", "TextParent2", "ListStatut", "ListPromotion", "ListAnnee", "TextDateInscription")
    For Each nom In champs
        Set c = ControleDisposition(f, CStr(nom))
        If Not c Is Nothing Then
            x = 12 + (i Mod 4) * (col + 8)
            y = 64 + (i \ 4) * 54
            c.Visible = True: c.Left = x: c.Top = y + 20
            c.Width = col: c.Height = 26: c.Font.Size = 10
            If c.Name = "TextAdresse" Then c.MultiLine = False
            Set l = ControleDisposition(f, "ProLibelle_" & c.Name)
            If l Is Nothing Then Set l = f.Controls.Add("Forms.Label.1", "ProLibelle_" & c.Name, True)
            With l
                .Visible = True: .Caption = LibelleChampPro(c.Name)
                If c.Name = "ListPromotion" Then .Caption = "Promotion"
                If c.Name = "ListStatut" Then .Caption = "Statut de l'apprenant"
                If c.Name = "TextDateNaissance" Then .Caption = "Naissance (jj/mm/aaaa)"
                .Left = x: .Top = y: .Width = col: .Height = 18
                .Font.Name = "Segoe UI": .Font.Size = 9: .Font.Bold = True
                .BackStyle = fmBackStyleTransparent: .ForeColor = RGB(71, 85, 105)
            End With
            If c.Top + c.Height > bas Then bas = c.Top + c.Height
            i = i + 1
        End If
    Next nom
    For Each c In f.Controls
        If c.Visible And TypeName(c) = "CommandButton" Then
            c.Left = 12 + (n Mod 4) * (col + 8)
            c.Top = bas + 18 + (n \ 4) * 34
            c.Width = col: c.Height = 26
            c.Font.Size = 9: c.WordWrap = True
            EtiqueterBoutonPro c: ColorerBoutonPro c
            n = n + 1
        End If
    Next c
    bas = bas + 18 + ((n + 3) \ 4) * 34 + 10
    f.Height = Application.Min(bas + (f.Height - f.InsideHeight), Application.Height - 48)
    f.ScrollHeight = bas
    If bas > f.InsideHeight Then
        f.ScrollBars = fmScrollBarsVertical
    Else
        f.ScrollBars = fmScrollBarsNone
    End If
    f.ScrollTop = 0: f.StartUpPosition = 0
    f.Left = Application.Left + (Application.Width - f.Width) / 2
    f.Top = Application.Top + (Application.Height - f.Height) / 2
End Sub
