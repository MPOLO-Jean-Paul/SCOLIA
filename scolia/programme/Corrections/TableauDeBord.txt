Private mChargement As Boolean


' ============================================================
' ITM COEUR DE DIEU - TABLEAU DE BORD SECRETARIAT
' USERFORM : UF_TableauDeBord
'
' SOURCES :
' APPRENANTS :
' A Matricule | B Nom | C Post-nom | D Prénom | H Promotion
'
' PAIEMENTS :
' D Matricule | F Promotion | G Mois | I Montant affecté
'
' MENSUALITE : 55 000 FC
' MOIS : Septembre à Mai
'
' CONTROLES A CREER EXACTEMENT :
'
' FILTRES
' ComboPromotionDashboard
' ComboMoisDashboard
' ComboNombreMoisLitige
'
' INDICATEURS
' TextTotalApprenants
' TextApprenantsPromotion
' TextRevenuTotal
' TextRevenuMois
' TextNombreLitiges
' TextMontantLitige
' TextTauxPaiement
' TextTauxAJour
'
' LISTE DETAILLEE
' ListApprenantsLitige
'
' BOUTONS
' CommandActualiserDashboard
' CommandFermerDashboard
'
' IMPORTANT :
' - "LITIGE" = mois partiellement payé.
' - "NON PAYE" = mois sans aucun paiement.
' - Le filtre "Nombre de mois en litige" compte uniquement les mois
'   partiellement payés, conformément au fonctionnement de PAIEMENTS.
' - Le taux de paiement financier =
'   montant affecté / montant théorique attendu.
' ============================================================

Private Const MONTANT_MENSUEL As Currency = 55000


Private Sub CommandFemerDashboard_Click()
Unload Me
End Sub

' ============================================================
' INITIALISATION
' ============================================================

Private Sub UserForm_Initialize()
    On Error GoTo EH
    mChargement = True

    ChargerPromotions
    ChargerMois
    ChargerNombreMoisLitige
    ConfigurerAffichage

    ComboPromotionDashboard.Value = "Toutes les promotions"
    ComboMoisDashboard.Value = "Tous les mois"
    ComboNombreMoisLitige.Value = "Tous"

    Set mActionsPro = ConfigurerFormulairePro(Me)
    mChargement = False
    ActualiserDashboard
    Exit Sub
EH:
    mChargement = False
    MsgBox "Ouverture du tableau de bord impossible : " & Err.Description, vbExclamation
End Sub

' ============================================================

    ChargerPromotionsValides ComboPromotionDashboard, True
End Sub


' ============================================================
' CHARGER LES MOIS
' ============================================================

Private Sub ChargerMois()

    Dim mois As Variant
    Dim m As Variant

    ComboMoisDashboard.Clear
    ComboMoisDashboard.AddItem "Tous les mois"

    mois = TableauMois()

    For Each m In mois
        ComboMoisDashboard.AddItem CStr(m)
    Next m

End Sub


' ============================================================
' CHARGER LE FILTRE NOMBRE DE MOIS EN LITIGE
' ============================================================

Private Sub ChargerNombreMoisLitige()

    Dim i As Integer

    ComboNombreMoisLitige.Clear
    ComboNombreMoisLitige.AddItem "Tous"

    For i = 1 To 9
        ComboNombreMoisLitige.AddItem CStr(i) & " mois"
    Next i

End Sub


' ============================================================
' EVENEMENTS DES FILTRES

Private Sub ComboPromotionDashboard_Change()

    If ComboPromotionDashboard.ListIndex >= 0 Then
        ActualiserDashboard
    End If

End Sub

Private Sub ComboMoisDashboard_Change()
    If mChargement Then Exit Sub

    If ComboMoisDashboard.ListIndex >= 0 Then
        ActualiserDashboard
    End If

End Sub

Private Sub ComboNombreMoisLitige_Change()
    If mChargement Then Exit Sub

    If ComboNombreMoisLitige.ListIndex >= 0 Then
        ChargerListeLitiges
    End If

End Sub


' ============================================================
' ACTUALISATION GENERALE
' ============================================================

Private Sub ActualiserDashboard()

    Dim promotion As String
    Dim mois As String

    Dim totalApprenants As Long
    Dim apprenantsPromotion As Long

    Dim revenuGlobal As Currency
    Dim revenuMois As Currency

    Dim nbLitiges As Long
    Dim montantLitige As Currency

    Dim tauxPaiement As Double
    Dim tauxAJour As Double

    promotion = Trim$(ComboPromotionDashboard.Value)
    mois = Trim$(ComboMoisDashboard.Value)

    totalApprenants = CompterApprenants("Toutes les promotions")
    apprenantsPromotion = CompterApprenants(promotion)

    revenuGlobal = CalculerRevenu(promotion, "Tous les mois")
    revenuMois = CalculerRevenu(promotion, mois)

    CalculerLitiges promotion, mois, nbLitiges, montantLitige

    tauxPaiement = CalculerTauxPaiement(promotion, mois)
    tauxAJour = CalculerTauxAJour(promotion, mois)

    TextTotalAp.Value = Format(totalApprenants, "#,##0")
    TextApprenantsPromotion.Value = Format(apprenantsPromotion, "#,##0")

    TextRevenuTotal.Value = Format(revenuGlobal, "#,##0") & " FC"
    TextRevenuMois.Value = Format(revenuMois, "#,##0") & " FC"

    TextNombreLitiges.Value = Format(nbLitiges, "#,##0")
    TextMontantLitige.Value = Format(montantLitige, "#,##0") & " FC"

    TextTauxPaiement.Value = Format(tauxPaiement, "0.00") & " %"
    TextTauxAjour.Value = Format(tauxAJour, "0.00") & " %"

    ChargerListeLitiges

End Sub


' ============================================================
' COMPTER LES APPRENANTS
' ============================================================

Private Function CompterApprenants(ByVal promotion As String) As Long

    Dim ws As Worksheet
    Dim derniereLigne As Long
    Dim i As Long
    Dim compteur As Long

    Set ws = ThisWorkbook.Worksheets("APPRENANTS")

    derniereLigne = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    compteur = 0

    If derniereLigne >= 5 Then

        For i = 5 To derniereLigne

            If Trim$(UCase$(CStr(ws.Cells(i, "A").Value))) <> "" Then

                If PromotionCorrespond( _
                    CStr(ws.Cells(i, "H").Value), promotion) Then

                    compteur = compteur + 1

                End If

            End If

        Next i

    End If

    CompterApprenants = compteur

End Function


' ============================================================
' REVENU
' ============================================================

Private Function CalculerRevenu( _
    ByVal promotion As String, _
    ByVal mois As String) As Currency

    Dim ws As Worksheet
    Dim derniereLigne As Long
    Dim i As Long
    Dim Total As Currency

    Set ws = ThisWorkbook.Worksheets("PAIEMENTS")

    derniereLigne = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    Total = 0

    If derniereLigne >= 5 Then

        For i = 5 To derniereLigne

            If PromotionCorrespond( _
                CStr(ws.Cells(i, "F").Value), promotion) _
            And MoisCorrespond( _
                CStr(ws.Cells(i, "G").Value), mois) Then

                If IsNumeric(ws.Cells(i, "I").Value) Then
                    Total = Total + CCur(ws.Cells(i, "I").Value)
                End If

            End If

        Next i

    End If

    CalculerRevenu = Total

End Function


' ============================================================
' LITIGES
'
' Un mois est LITIGE si :
' payé > 0 ET payé < 55 000.
'
' Le montant du litige est le reste à solder :
' 55 000 - total déjà affecté au mois.
' ============================================================

Private Sub CalculerLitiges( _
    ByVal promotion As String, _
    ByVal filtreMois As String, _
    ByRef nombreApprenants As Long, _
    ByRef montantTotal As Currency)

    Dim ws As Worksheet
    Dim derniereLigne As Long
    Dim i As Long

    Dim matricule As String
    Dim promoApprenant As String
    Dim mois As Variant
    Dim m As Variant

    Dim totalMois As Currency
    Dim d As Object

    Set ws = ThisWorkbook.Worksheets("APPRENANTS")
    Set d = CreateObject("Scripting.Dictionary")

    nombreApprenants = 0
    montantTotal = 0

    derniereLigne = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    mois = TableauMois()

    If derniereLigne < 5 Then Exit Sub

    For i = 5 To derniereLigne

        matricule = Trim$(UCase$(CStr(ws.Cells(i, "A").Value)))
        promoApprenant = Trim$(CStr(ws.Cells(i, "H").Value))

        If matricule <> "" _
        And PromotionCorrespond(promoApprenant, promotion) Then

            For Each m In mois

                If MoisCorrespond(CStr(m), filtreMois) Then

                    totalMois = TotalPayePourMoisDashboard( _
                        matricule, CStr(m))

                    If totalMois > 0 _
                    And totalMois < MONTANT_MENSUEL Then

                        montantTotal = montantTotal + _
                            (MONTANT_MENSUEL - totalMois)

                        If Not d.Exists(matricule) Then
                            d.Add matricule, True
                        End If

                    End If

                End If

            Next m

        End If

    Next i

    nombreApprenants = d.Count

End Sub


' ============================================================
' TAUX DE PAIEMENT FINANCIER
'
' Tous les mois :
' payé / (effectif * 55 000 * 9)
'
' Un mois :
' payé pour ce mois / (effectif * 55 000)
' ============================================================

Private Function CalculerTauxPaiement( _
    ByVal promotion As String, _
    ByVal mois As String) As Double

    Dim effectif As Long
    Dim attendu As Currency
    Dim paye As Currency
    Dim nombreMois As Long

    effectif = CompterApprenants(promotion)

    If effectif = 0 Then
        CalculerTauxPaiement = 0
        Exit Function
    End If

    If mois = "" Or _
       StrComp(mois, "Tous les mois", vbTextCompare) = 0 Then

        nombreMois = 9

    Else

        nombreMois = 1

    End If

    attendu = CCur(effectif) * MONTANT_MENSUEL * nombreMois
    paye = CalculerRevenu(promotion, mois)

    If attendu <= 0 Then
        CalculerTauxPaiement = 0
    Else
        CalculerTauxPaiement = (paye / attendu) * 100
    End If

End Function


' ============================================================
' TAUX D'APPRENANTS A JOUR
'
' Pour un mois :
' apprenants ayant payé 55 000 / effectif
'
' Tous les mois :
' apprenants ayant soldé les 9 mois / effectif
' ============================================================

Private Function CalculerTauxAJour( _
    ByVal promotion As String, _
    ByVal filtreMois As String) As Double

    Dim ws As Worksheet
    Dim derniereLigne As Long
    Dim i As Long

    Dim matricule As String
    Dim promoApprenant As String

    Dim effectif As Long
    Dim nbAJour As Long

    Dim mois As Variant
    Dim m As Variant
    Dim estAJour As Boolean

    Set ws = ThisWorkbook.Worksheets("APPRENANTS")

    derniereLigne = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    effectif = 0
    nbAJour = 0

    mois = TableauMois()

    If derniereLigne < 5 Then
        CalculerTauxAJour = 0
        Exit Function
    End If

    For i = 5 To derniereLigne

        matricule = Trim$(UCase$(CStr(ws.Cells(i, "A").Value)))
        promoApprenant = Trim$(CStr(ws.Cells(i, "H").Value))

        If matricule <> "" _
        And PromotionCorrespond(promoApprenant, promotion) Then

            effectif = effectif + 1

            If filtreMois = "" Or _
               StrComp(filtreMois, "Tous les mois", vbTextCompare) = 0 Then

                estAJour = True

                For Each m In mois

                    If TotalPayePourMoisDashboard( _
                        matricule, CStr(m)) < MONTANT_MENSUEL Then

                        estAJour = False
                        Exit For

                    End If

                Next m

            Else

                estAJour = _
                    (TotalPayePourMoisDashboard( _
                    matricule, filtreMois) >= MONTANT_MENSUEL)

            End If

            If estAJour Then
                nbAJour = nbAJour + 1
            End If

        End If

    Next i

    If effectif = 0 Then
        CalculerTauxAJour = 0
    Else
        CalculerTauxAJour = (nbAJour / effectif) * 100
    End If

End Function


' ============================================================
' LISTE DES APPRENANTS AYANT DES LITIGES
'
' Colonnes :
' Matricule | Nom complet | Promotion | Nb mois litige | Dette litige
' ============================================================

Private Sub ChargerListeLitiges()

    Dim ws As Worksheet
    Dim derniereLigne As Long
    Dim i As Long
    Dim indexListe As Long

    Dim matricule As String
    Dim nomComplet As String
    Dim promotion As String
    Dim filtrePromotion As String
    Dim filtreMois As String

    Dim mois As Variant
    Dim m As Variant

    Dim totalMois As Currency
    Dim detteLitige As Currency
    Dim nbMoisLitige As Long
    Dim filtreNombre As Long

    Set ws = ThisWorkbook.Worksheets("APPRENANTS")

    ListApprenantsLitige.Clear

    filtrePromotion = Trim$(ComboPromotionDashboard.Value)
    filtreMois = Trim$(ComboMoisDashboard.Value)
    filtreNombre = NombreMoisLitigeSelectionne()

    derniereLigne = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    If derniereLigne < 5 Then Exit Sub

    mois = TableauMois()

    For i = 5 To derniereLigne

        matricule = Trim$(UCase$(CStr(ws.Cells(i, "A").Value)))
        promotion = Trim$(CStr(ws.Cells(i, "H").Value))

        If matricule <> "" _
        And PromotionCorrespond(promotion, filtrePromotion) Then

            nbMoisLitige = 0
            detteLitige = 0

            For Each m In mois

                If MoisCorrespond(CStr(m), filtreMois) Then

                    totalMois = TotalPayePourMoisDashboard( _
                        matricule, CStr(m))

                    If totalMois > 0 _
                    And totalMois < MONTANT_MENSUEL Then

                        nbMoisLitige = nbMoisLitige + 1

                        detteLitige = detteLitige + _
                            (MONTANT_MENSUEL - totalMois)

                    End If

                End If

            Next m

            If nbMoisLitige > 0 Then

                If filtreNombre = 0 _
                Or nbMoisLitige = filtreNombre Then

                    nomComplet = Trim$( _
                        UCase$(CStr(ws.Cells(i, "B").Value)) & " " & _
                        UCase$(CStr(ws.Cells(i, "C").Value)) & " " & _
                        UCase$(CStr(ws.Cells(i, "D").Value)))

                    ListApprenantsLitige.AddItem matricule

                    indexListe = _
                        ListApprenantsLitige.ListCount - 1

                    ListApprenantsLitige.List( _
                        indexListe, 1) = nomComplet

                    ListApprenantsLitige.List( _
                        indexListe, 2) = promotion

                    ListApprenantsLitige.List( _
                        indexListe, 3) = nbMoisLitige

                    ListApprenantsLitige.List( _
                        indexListe, 4) = _
                        Format(detteLitige, "#,##0") & " FC"

                End If

            End If

        End If

    Next i

End Sub


' ============================================================
' NOMBRE DE MOIS EN LITIGE SELECTIONNE
' 0 = TOUS
' ============================================================

Private Function NombreMoisLitigeSelectionne() As Long

    Dim valeur As String

    valeur = Trim$(ComboNombreMoisLitige.Value)

    If valeur = "" _
    Or StrComp(valeur, "Tous", vbTextCompare) = 0 Then

        NombreMoisLitigeSelectionne = 0

    Else

        NombreMoisLitigeSelectionne = Val(valeur)

    End If

End Function


' ============================================================
' TOTAL PAYE PAR APPRENANT ET PAR MOIS
' ============================================================

Private Function TotalPayePourMoisDashboard( _
    ByVal matricule As String, _
    ByVal mois As String) As Currency

    Dim ws As Worksheet
    Dim derniereLigne As Long
    Dim i As Long
    Dim Total As Currency

    Set ws = ThisWorkbook.Worksheets("PAIEMENTS")

    derniereLigne = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row

    Total = 0

    If derniereLigne >= 5 Then

        For i = 5 To derniereLigne

            If StrComp( _
                Trim$(UCase$(CStr(ws.Cells(i, "D").Value))), _
                Trim$(matricule), _
                vbTextCompare) = 0 _
            And StrComp( _
                Trim$(CStr(ws.Cells(i, "G").Value)), _
                Trim$(mois), _
                vbTextCompare) = 0 Then

                If IsNumeric(ws.Cells(i, "I").Value) Then
                    Total = Total + CCur(ws.Cells(i, "I").Value)
                End If

            End If

        Next i

    End If

    TotalPayePourMoisDashboard = Total

End Function


' ============================================================
' CORRESPONDANCE PROMOTION
' ============================================================

Private Function PromotionCorrespond(ByVal promotionLigne As String, ByVal promotionFiltre As String) As Boolean
    PromotionCorrespond = ITMPromotionCorrespond(promotionLigne, promotionFiltre)
End Function
' ============================================================
' CORRESPONDANCE MOIS
' ============================================================

Private Function MoisCorrespond( _
    ByVal moisLigne As String, _
    ByVal filtre As String) As Boolean

    If filtre = "" _
    Or StrComp(filtre, _
        "Tous les mois", vbTextCompare) = 0 Then

        MoisCorrespond = True

    Else

        MoisCorrespond = _
            (StrComp( _
            Trim$(moisLigne), _
            Trim$(filtre), _
            vbTextCompare) = 0)

    End If

End Function


' ============================================================
' TABLEAU DES MOIS
' ============================================================

Private Function TableauMois() As Variant

    TableauMois = Array( _
        "Septembre", _
        "Octobre", _
        "Novembre", _
        "Décembre", _
        "Janvier", _
        "Février", _
        "Mars", _
        "Avril", _
        "Mai")

End Function

' BOUTON ACTUALISER
' ============================================================

    On Error GoTo EH

    ITMRestaurerChoix ComboNombreMoisLitige, nombre, "Tous"
    mChargement = False
    ActualiserDashboard
    Exit Sub
EH:
    mChargement = False
    MsgBox "Actualisation impossible : " & Err.Description, vbExclamation
End Sub
