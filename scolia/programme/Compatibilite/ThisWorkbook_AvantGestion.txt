Option Explicit
Private Sub Workbook_BeforeSave(ByVal SaveAsUI As Boolean, Cancel As Boolean)
    Dim eventsAvant As Boolean, descriptionErreur As String
    If Me.Path = "" Then Exit Sub
    eventsAvant = Application.EnableEvents
    On Error GoTo EH
    Application.EnableEvents = False
    Call ITMSauvegarde
    Application.EnableEvents = eventsAvant
    Exit Sub
EH:
    descriptionErreur = Err.Description
    Application.EnableEvents = eventsAvant
    Cancel = True
    MsgBox "Sauvegarde de securite impossible : " & descriptionErreur & vbCrLf & "L'enregistrement a ete annule. Corrigez l'acces au dossier Sauvegardes.", vbCritical
End Sub
