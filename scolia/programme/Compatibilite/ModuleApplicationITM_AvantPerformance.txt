Private Function AJ(ByVal valeur As Variant) As String
    Dim s As String, i As Long, c As String, code As Long
    If IsError(valeur) Or IsNull(valeur) Or IsEmpty(valeur) Then AJ = "null": Exit Function
    If VarType(valeur) = vbBoolean Then
        AJ = IIf(valeur, "true", "false"): Exit Function
    End If
    If VarType(valeur) <> vbString And IsNumeric(valeur) Then
        AJ = Trim$(Str$(valeur))
        If Left$(AJ, 1) = "." Then AJ = "0" & AJ
        If Left$(AJ, 2) = "-." Then AJ = "-0" & Mid$(AJ, 2)
        Exit Function
    End If
    s = CStr(valeur)
    AJ = Chr$(34)
    For i = 1 To Len(s)
        c = Mid$(s, i, 1): code = AscW(c)
        Select Case code
            Case 34: AJ = AJ & "\" & Chr$(34)
            Case 92: AJ = AJ & "\\"
            Case 0 To 31: AJ = AJ & "\u" & Right$("0000" & Hex$(code), 4)
            Case Else: AJ = AJ & c
        End Select
    Next i
    AJ = AJ & Chr$(34)
End Function

Private Function ADate(ByVal valeur As Variant) As String
    If IsDate(valeur) Then ADate = Format$(CDate(valeur), "dd/mm/yyyy")
End Function

Public Function ITMAppSnapshot() As String
    Dim livres As Collection, livre As Workbook
    Dim d As Object, w As Worksheet, r As Long, i As Long, data As Variant
    Dim s As String, first As Boolean, mat As String, eligible As Boolean, noms As Variant
    Dim k As Variant, part As Variant, v As Variant, vus As Object
    Dim dates As Variant, last As Long, hasValues As Boolean
    Set d = ITMChargerDonnees()
    s = "{""learners"":[": first = True
    Set livres = ITMClasseurs()
    For Each livre In livres
    Set w = livre.Worksheets("APPRENANTS")
    noms = Split("Matricule|Nom|PostNom|Prenom|Sexe|Naissance|Lieu|Promotion|Tuteur|Contact1|Contact2|Inscription|Annee|Statut|Adresse", "|")
    last = ITMDerniereLigne(w)
    If last >= 5 Then
    data = w.Range(w.Cells(5, 1), w.Cells(last, 15)).Value2
    dates = w.Range(w.Cells(5, 1), w.Cells(last, 15)).Value
    For r = 1 To UBound(data, 1)
        hasValues = False
        For i = 1 To 15
            If Not IsEmpty(data(r, i)) Then hasValues = True: Exit For
        Next i
        If hasValues Then
            If Not first Then s = s & ","
            first = False: mat = ITMCleTexte(data(r, 1))
            eligible = d("personnes").Exists(mat)
            s = s & "{""Source"":" & AJ(livre.Name) & ",""Row"":" & (r + 4) & ",""Eligible"":" & AJ(eligible)
            For i = 1 To 15
                v = data(r, i)
                If i = 6 Or i = 12 Then v = ADate(dates(r, i))
                s = s & "," & AJ(noms(i - 1)) & ":" & AJ(v)
            Next i
            s = s & "}"
        End If
    Next r
    End If
    Next livre
    s = s & "],""payments"":[": first = True
    For Each livre In livres
    Set w = livre.Worksheets("PAIEMENTS")
    noms = Split("ID|Date|Numero|Matricule|NomComplet|Promotion|Mois|Du|Verse|Reste|Statut|TotalPaye|TotalAnnuel|ResteAnnuel", "|")
    last = ITMDerniereLigne(w)
    If last >= 5 Then
    data = w.Range(w.Cells(5, 1), w.Cells(last, 14)).Value2
    dates = w.Range(w.Cells(5, 1), w.Cells(last, 14)).Value
    For r = 1 To UBound(data, 1)
        hasValues = False
        For i = 1 To 14
            If Not IsEmpty(data(r, i)) Then hasValues = True: Exit For
        Next i
        If hasValues Then
            If Not first Then s = s & ","
            first = False: s = s & "{""Source"":" & AJ(livre.Name) & ",""Row"":" & (r + 4)
            For i = 1 To 14
                v = data(r, i)
                If i = 2 Then v = ADate(dates(r, i))
                s = s & "," & AJ(noms(i - 1)) & ":" & AJ(v)
            Next i
            s = s & "}"
        End If
    Next r
    End If
    Next livre
    s = s & "],""promotions"":[": first = True
    Set vus = CreateObject("Scripting.Dictionary")
    Set w = ThisWorkbook.Worksheets("PROMOTIONS")
    For r = 5 To w.Cells(w.Rows.Count, 1).End(xlUp).Row
        mat = ITMPromotion(w.Cells(r, 1).Value2)
        If d("promotions").Exists(mat) And Not vus.Exists(mat) Then
            vus(mat) = True
            If Not first Then s = s & ","
            first = False
            s = s & "{""Code"":" & AJ(mat) & ",""Designation"":" & AJ(w.Cells(r, 2).Value2) & ",""Annee"":" & AJ(w.Cells(r, 5).Value2) & ",""Mensuel"":" & AJ(w.Cells(r, 6).Value2) & ",""Annuel"":" & AJ(w.Cells(r, 7).Value2) & ",""Active"":" & AJ(ITMCleTexte(w.Cells(r, 8).Value2) = "ACTIVE") & "}"
        End If
    Next r
    s = s & "],""months"":[": first = True
    For Each k In d("mois").Keys
        If Not first Then s = s & ","
        first = False: s = s & AJ(d("mois")(k))
    Next k
    s = s & "],""balances"":[": first = True
    For Each k In d("sommes").Keys
        If Not first Then s = s & ","
        first = False: part = Split(CStr(k), "|")
        s = s & "{""Matricule"":" & AJ(part(0)) & ",""Mois"":" & AJ(part(1)) & ",""Verse"":" & AJ(d("sommes")(k)) & "}"
    Next k
    s = s & "],""issues"":[": first = True
    For Each v In d("issues")
        If Not first Then s = s & ","
        first = False
        s = s & "{""Feuille"":" & AJ(v(0)) & ",""Ligne"":" & AJ(v(1)) & ",""Matricule"":" & AJ(v(2)) & ",""Numero"":" & AJ(v(3)) & ",""Observation"":" & AJ(v(4)) & "}"
    Next v
    ITMAppSnapshot = s & "],""closed"":" & AJ(ITMAnneeFermee()) & "}"
End Function

Private Sub AValiderIdentite(ByVal nom As String, ByVal post As String, ByVal prenom As String, ByVal sexe As String, ByVal naissance As String, ByVal lieu As String, ByVal tuteur As String, ByVal contact1 As String, ByVal contact2 As String, ByVal promo As String, ByVal annee As String, ByVal statut As String, Optional ByVal requireActive As Boolean = True)
    Dim d As Date, expression As Object, w As Worksheet, r As Long, trouve As Boolean
    Set expression = CreateObject("VBScript.RegExp")
    expression.Pattern = "^[A-Za-zÀ-ÖØ-öø-ÿ '’-]+$"
    If Trim$(nom) = "" Or Not expression.Test(Trim$(nom)) Then Err.Raise vbObjectError + 1000, , "Nom obligatoire : lettres uniquement."
    If Trim$(post) = "" Or Not expression.Test(Trim$(post)) Then Err.Raise vbObjectError + 1001, , "Post-nom obligatoire : lettres uniquement."
    If Trim$(prenom) = "" Or Not expression.Test(Trim$(prenom)) Then Err.Raise vbObjectError + 1002, , "Prénom obligatoire : lettres uniquement."
    If sexe <> "M" And sexe <> "F" Then Err.Raise vbObjectError + 1003, , "Choisissez le sexe."
    If Not ITMLireDate(naissance, d) Then Err.Raise vbObjectError + 1004, , "Date de naissance incorrecte (jj/mm/aaaa)."
    If Trim$(lieu) = "" Then Err.Raise vbObjectError + 1005, , "Lieu de naissance obligatoire."
    If Not ContactValide(contact1, True) Or Not ContactValide(contact2, False) Then Err.Raise vbObjectError + 1006, , "Contacts : 10 chiffres, le second peut rester vide."
    If statut <> "Actif" And statut <> "Suspendu" And statut <> "Sorti" Then Err.Raise vbObjectError + 1007, , "Statut non reconnu."
    Set w = ThisWorkbook.Worksheets("PROMOTIONS")
    For r = 5 To w.Cells(w.Rows.Count, 1).End(xlUp).Row
        If ITMPromotion(w.Cells(r, 1).Value2) = ITMPromotion(promo) And (Not requireActive Or ITMCleTexte(w.Cells(r, 8).Value2) = "ACTIVE") Then
            If Replace(CStr(w.Cells(r, 5).Value2), " ", "") = Replace(annee, " ", "") Then trouve = True
        End If
    Next r
    If Not trouve Or ITMFraisMensuels(promo) <= 0 Or ITMFraisAnnuels(promo) < ITMFraisMensuels(promo) Then Err.Raise vbObjectError + 1008, , "Promotion active et année scolaire compatibles obligatoires."
End Sub

Private Function AMatricule(ByVal promo As String) As String
    Dim alphabet As String, a As Long, b As Long, n As Long, code As String, w As Worksheet, nom As Variant, col As Long, r As Long, pris As Object
    Dim livres As Collection, livre As Workbook
    Set livres = ITMClasseurs()
    promo = ITMPromotion(promo)
    If Len(promo) <> 4 Then Err.Raise vbObjectError + 1009, , "Le code promotion doit avoir 4 caractères."
    Set pris = CreateObject("Scripting.Dictionary")
    For Each livre In livres
    For Each nom In Array("APPRENANTS", "PAIEMENTS")
        Set w = livre.Worksheets(CStr(nom)): col = IIf(nom = "APPRENANTS", 1, 4)
        For r = 5 To ITMDerniereLigne(w)
            code = ITMCleTexte(w.Cells(r, col).Value2)
            If code <> "" Then pris(code) = True
        Next r
    Next nom
    Next livre
    For n = 1 To 99
        code = promo & Format$(n, "00")
        If Not pris.Exists(code) Then AMatricule = code: Exit Function
    Next n
    alphabet = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    For a = 1 To 36
        For b = 1 To 36
            code = Mid$(alphabet, a, 1) & Mid$(alphabet, b, 1)
            If Not code Like "##" Then
                code = promo & code
                If Not pris.Exists(code) Then AMatricule = code: Exit Function
            End If
        Next b
    Next a
    Err.Raise vbObjectError + 1010, , "Les matricules de cette promotion sont épuisés."
End Function

Private Function AIdentiteCle(ByVal valeur As Variant) As String
    Dim texte As String
    If IsError(valeur) Or IsNull(valeur) Or IsEmpty(valeur) Then Exit Function
    texte = Replace(Replace(Replace(Replace(CStr(valeur), vbTab, " "), vbCr, " "), vbLf, " "), ChrW(8239), " ")
    AIdentiteCle = ITMCleTexte(texte)
End Function

Public Function ITMAppDossier(ByVal matricule As String, ByVal nom As String, ByVal post As String, ByVal prenom As String, ByVal sexe As String, ByVal naissance As String, ByVal lieu As String, ByVal tuteur As String, ByVal contact1 As String, ByVal contact2 As String, ByVal promo As String, ByVal annee As String, ByVal statut As String, ByVal adresse As String) As String
    Dim w As Worksheet, r As Long, i As Long, row As Long, d As Date, vieux As Variant
    Dim actif As Boolean, nouveau As Boolean, erreur As String, doublon As Long
    Dim avant As String, oldPayments As Variant, pw As Worksheet, paymentLast As Long, nr As Long
    Dim livres As Collection, livre As Workbook
    Dim controle As Worksheet, globalData As Object, globalKey As Variant
    On Error GoTo EH
    contact1 = Trim$(contact1): contact2 = Trim$(contact2)
    AValiderIdentite nom, post, prenom, sexe, naissance, lieu, tuteur, contact1, contact2, promo, annee, statut, (Trim$(matricule) = "")
    ITMLireDate naissance, d
    Set w = ThisWorkbook.Worksheets("APPRENANTS")
    nouveau = Trim$(matricule) = ""
    Set livres = ITMClasseurs()
    For Each livre In livres
        Set controle = livre.Worksheets("APPRENANTS")
    For i = 5 To ITMDerniereLigne(controle)
            If nouveau Or ITMCleTexte(controle.Cells(i, 1).Value2) <> ITMCleTexte(matricule) Then
                If AIdentiteCle(controle.Cells(i, 2).Value2) = AIdentiteCle(nom) And AIdentiteCle(controle.Cells(i, 3).Value2) = AIdentiteCle(post) And AIdentiteCle(controle.Cells(i, 4).Value2) = AIdentiteCle(prenom) And ITMPromotion(controle.Cells(i, 8).Value2) = ITMPromotion(promo) Then
                    Err.Raise vbObjectError + 1011, , "Cet apprenant existe deja dans cette promotion. Matricule : " & CStr(controle.Cells(i, 1).Value2) & ". Consultez et modifiez son dossier existant."
                End If
            End If
    Next i
    Next livre
    If nouveau Then
        matricule = AMatricule(promo)
    Else
        For i = 5 To ITMDerniereLigne(w)
            If ITMCleTexte(w.Cells(i, 1).Value2) = ITMCleTexte(matricule) Then row = i: doublon = doublon + 1
        Next i
        If doublon <> 1 Then Err.Raise vbObjectError + 1012, , "Dossier absent ou matricule en double : modification bloquée."
        ITMPECInitialiser
        If ITMPECExiste(matricule, CStr(w.Cells(row, 13).Value2)) Then
            If ITMPromotion(w.Cells(row, 8).Value2) <> ITMPromotion(promo) Or ITMPECAnnee(CStr(w.Cells(row, 13).Value2)) <> ITMPECAnnee(annee) Then Err.Raise vbObjectError + 1197, , "Prise en charge enregistrée : année et promotion conservées. Utilisez la clôture pour changer d'année."
        End If
        vieux = w.Range(w.Cells(row, 1), w.Cells(row, 15)).Formula
        Set pw = ThisWorkbook.Worksheets("PAIEMENTS"): paymentLast = ITMDerniereLigne(pw)
        If paymentLast >= 5 Then oldPayments = pw.Range(pw.Cells(5, 1), pw.Cells(paymentLast, 14)).Formula
        For nr = 5 To paymentLast
            If ITMCleTexte(pw.Cells(nr, 4).Value2) = ITMCleTexte(matricule) Then
                If ITMPromotion(promo) <> ITMPromotion(w.Cells(row, 8).Value2) Or annee <> CStr(w.Cells(row, 13).Value2) Then Err.Raise vbObjectError + 1030, , "Des paiements existent : promotion et annee conservees dans ce dossier."
            End If
        Next nr
        Set globalData = ITMChargerDonnees()
        For Each globalKey In globalData("sommes").Keys
            If Split(CStr(globalKey), "|")(0) = ITMCleTexte(matricule) Then
                If ITMPromotion(promo) <> ITMPromotion(w.Cells(row, 8).Value2) Or annee <> CStr(w.Cells(row, 13).Value2) Then Err.Raise vbObjectError + 1030, , "Des paiements existent dans le dossier de donnees : promotion et annee conservees."
            End If
        Next globalKey
        avant = ITMJsonLigne(w, row, 15)
    End If
    ITMDebutSaisie w: actif = True: ITMJournalDebut
    If nouveau Then row = ITMReserverLigne(w)
    w.Range(w.Cells(row, 1), w.Cells(row, 15)).NumberFormat = "@"
    w.Cells(row, 1).Value2 = matricule
    w.Cells(row, 2).Value2 = UCase$(Trim$(nom)): w.Cells(row, 3).Value2 = UCase$(Trim$(post)): w.Cells(row, 4).Value2 = UCase$(Trim$(prenom))
    w.Cells(row, 5).Value2 = sexe: w.Cells(row, 6).NumberFormat = "dd/mm/yyyy": w.Cells(row, 6).Value = d
    w.Cells(row, 7).Value2 = Trim$(lieu): w.Cells(row, 8).Value2 = ITMPromotion(promo): w.Cells(row, 9).Value2 = Trim$(tuteur)
    w.Cells(row, 10).Value2 = contact1: w.Cells(row, 11).Value2 = contact2
    If nouveau Then w.Cells(row, 12).Value = Date
    w.Cells(row, 12).NumberFormat = "dd/mm/yyyy"
    w.Cells(row, 13).Value2 = annee: w.Cells(row, 14).Value2 = statut: w.Cells(row, 15).Value2 = Left$(Trim$(adresse), 250)
    If Not nouveau And paymentLast >= 5 Then
        For nr = 5 To paymentLast
            If ITMCleTexte(pw.Cells(nr, 4).Value2) = ITMCleTexte(matricule) Then
                pw.Cells(nr, 5).NumberFormat = "@"
                pw.Cells(nr, 5).Value2 = UCase$(Trim$(nom) & " " & Trim$(post) & " " & Trim$(prenom))
            End If
        Next nr
    End If
    ITMJournalEcrire IIf(nouveau, "INSCRIPTION", "MODIFICATION_DOSSIER"), matricule, "", avant, ITMJsonLigne(w, row, 15), "Identite, contacts, adresse et statut"
    ITMValiderSaisie: actif = False: ITMJournalConfirmer
    mDerniereErreur = ""
    ITMAppDossier = "{""ok"":true,""matricule"":" & AJ(matricule) & ",""nouveau"":" & AJ(nouveau) & "}"
    Exit Function
EH:
    erreur = Err.Description
    Err.Clear
    On Error Resume Next
    If actif Then
        If Not nouveau And row > 0 Then w.Range(w.Cells(row, 1), w.Cells(row, 15)).Formula = vieux
        If Not nouveau And paymentLast >= 5 Then pw.Range(pw.Cells(5, 1), pw.Cells(paymentLast, 14)).Formula = oldPayments
        ITMCollecterErreurAnnulation erreur
        ITMJournalAnnuler
        ITMCollecterErreurAnnulation erreur
        ITMAnnulerSaisie
        ITMCollecterErreurAnnulation erreur
    End If
    On Error GoTo 0
    mDerniereErreur = erreur
    ITMAppDossier = "{""ok"":false,""error"":" & AJ(erreur) & "}"
End Function

Public Function ITMAppSupprimerApprenant(ByVal matricule As String, ByVal motif As String) As String
    Dim livres As Collection, livre As Workbook, w As Worksheet, pw As Worksheet
    Dim i As Long, row As Long, n As Long, last As Long, ancien As Variant
    Dim avant As String, erreur As String, actif As Boolean, nbPaiements As Long
    On Error GoTo EH
    ITMVerifierAnneeOuverte
    matricule = ITMCleTexte(matricule): motif = Trim$(motif)
    If matricule = "" Then Err.Raise vbObjectError + 1040, , "Selectionnez un apprenant."
    If Len(motif) < 5 Then Err.Raise vbObjectError + 1043, , "Le motif de suppression est obligatoire (5 caracteres minimum)."
    Set livres = ITMClasseurs()
    For Each livre In livres
        Set pw = livre.Worksheets("APPRENANTS")
        For i = 5 To ITMDerniereLigne(pw)
            If ITMCleTexte(pw.Cells(i, 1).Value2) = matricule Then
                ITMPECInitialiser
                If ITMPECExiste(matricule, CStr(pw.Cells(i, 13).Value2)) Then Err.Raise vbObjectError + 1198, , "Historique de prise en charge : suppression interdite. Utilisez Sorti ou Suspendu."
                n = n + 1
                If livre Is ThisWorkbook Then Set w = pw: row = i
            End If
        Next i
        Set pw = livre.Worksheets("PAIEMENTS")
        For i = 5 To ITMDerniereLigne(pw)
            If ITMCleTexte(pw.Cells(i, 4).Value2) = matricule Then nbPaiements = nbPaiements + 1
        Next i
    Next livre
    If n <> 1 Or row < 5 Then Err.Raise vbObjectError + 1042, , "Dossier absent, en double ou situe dans un autre classeur."
    last = ITMDerniereLigne(w)
    ancien = w.Range(w.Cells(5, 1), w.Cells(last, 15)).Formula
    avant = ITMJsonLigne(w, row, 15)
    ITMDebutSaisie w: actif = True: ITMJournalDebut
    If row < last Then w.Range(w.Cells(row, 1), w.Cells(last - 1, 15)).Formula = w.Range(w.Cells(row + 1, 1), w.Cells(last, 15)).Formula
    w.Range(w.Cells(last, 1), w.Cells(last, 15)).ClearContents
    w.ListObjects(1).Resize w.Range(w.Cells(4, 1), w.Cells(Application.Max(5, last - 1), 15))
    If nbPaiements > 0 Then motif = motif & " | " & CStr(nbPaiements) & " paiement(s) conserve(s) dans l'historique comptable."
    ITMJournalEcrire "SUPPRESSION_APPRENANT", matricule, "", avant, "", motif
    ITMValiderSaisie: actif = False: ITMJournalConfirmer
    ITMAppSupprimerApprenant = "{""ok"":true,""matricule"":" & AJ(matricule) & ",""paiementsConserves"":" & CStr(nbPaiements) & "}"
    Exit Function
EH:
    erreur = Err.Description
    On Error Resume Next
    If actif Then
        w.Range(w.Cells(5, 1), w.Cells(last, 15)).Formula = ancien
        w.ListObjects(1).Resize w.Range(w.Cells(4, 1), w.Cells(Application.Max(5, last), 15))
        ITMCollecterErreurAnnulation erreur
        ITMJournalAnnuler
    End If
    On Error GoTo 0
    ITMAppSupprimerApprenant = "{""ok"":false,""error"":" & AJ(erreur) & "}"
End Function

Public Function ITMAppFusionnerApprenants(ByVal matriculeConserver As String, ByVal matriculeFusion As String, ByVal motif As String) As String
    Dim livres As Collection, livre As Workbook, w As Worksheet, pw As Worksheet
    Dim keepW As Worksheet, mergeW As Worksheet, keepRow As Long, mergeRow As Long, keepCount As Long, mergeCount As Long
    Dim keepBook As Workbook, mergeBook As Workbook, i As Long, r As Long, lastL As Long, lastP As Long
    Dim books As Collection, oldLearners As Collection, oldPayments As Collection, learnerLasts As Collection, paymentLasts As Collection
    Dim avant As String, apres As String, erreur As String, savedCount As Long, j As Long, nbPaiements As Long
    Dim oldEvents As Boolean, oldScreen As Boolean, oldCalc As XlCalculation, started As Boolean, stateCaptured As Boolean
    On Error GoTo EH
    ITMVerifierAnneeOuverte
    matriculeConserver = ITMCleTexte(matriculeConserver): matriculeFusion = ITMCleTexte(matriculeFusion): motif = Trim$(motif)
    If matriculeConserver = "" Or matriculeFusion = "" Or matriculeConserver = matriculeFusion Then Err.Raise vbObjectError + 1210, , "Selectionnez deux matricules differents."
    If Len(motif) < 5 Then Err.Raise vbObjectError + 1211, , "Le motif de fusion est obligatoire (5 caracteres minimum)."
    Set livres = ITMClasseurs()
    For Each livre In livres
        Set w = livre.Worksheets("APPRENANTS")
        For i = 5 To ITMDerniereLigne(w)
            If ITMCleTexte(w.Cells(i, 1).Value2) = matriculeConserver Then keepCount = keepCount + 1: Set keepW = w: keepRow = i: Set keepBook = livre
            If ITMCleTexte(w.Cells(i, 1).Value2) = matriculeFusion Then mergeCount = mergeCount + 1: Set mergeW = w: mergeRow = i: Set mergeBook = livre
        Next i
    Next livre
    If keepCount <> 1 Or mergeCount <> 1 Then Err.Raise vbObjectError + 1212, , "Un dossier est absent ou un matricule n'est pas unique."
    If ITMCleTexte(keepW.Cells(keepRow, 2).Value2) <> ITMCleTexte(mergeW.Cells(mergeRow, 2).Value2) _
       Or ITMCleTexte(keepW.Cells(keepRow, 3).Value2) <> ITMCleTexte(mergeW.Cells(mergeRow, 3).Value2) _
       Or ITMCleTexte(keepW.Cells(keepRow, 4).Value2) <> ITMCleTexte(mergeW.Cells(mergeRow, 4).Value2) _
       Or ITMCleTexte(keepW.Cells(keepRow, 8).Value2) <> ITMCleTexte(mergeW.Cells(mergeRow, 8).Value2) Then
        Err.Raise vbObjectError + 1213, , "Fusion refusee : l'identite ou la promotion des deux dossiers differe."
    End If
    ITMPECInitialiser
    If ITMPECExiste(matriculeFusion, CStr(mergeW.Cells(mergeRow, 13).Value2)) Then Err.Raise vbObjectError + 1214, , "Le dossier a fusionner possede un historique de prise en charge. Traitez-le avant la fusion."
    avant = ITMJsonLigne(keepW, keepRow, 15) & vbCrLf & ITMJsonLigne(mergeW, mergeRow, 15)
    Set books = New Collection: Set oldLearners = New Collection: Set oldPayments = New Collection
    Set learnerLasts = New Collection: Set paymentLasts = New Collection
    oldEvents = Application.EnableEvents: oldScreen = Application.ScreenUpdating: oldCalc = Application.Calculation: stateCaptured = True
    Application.EnableEvents = False: Application.ScreenUpdating = False: Application.Calculation = xlCalculationManual
    For Each livre In livres
        Set w = livre.Worksheets("APPRENANTS"): Set pw = livre.Worksheets("PAIEMENTS")
        lastL = ITMDerniereLigne(w): lastP = ITMDerniereLigne(pw)
        books.Add livre: learnerLasts.Add lastL: paymentLasts.Add lastP
        If lastL >= 5 Then oldLearners.Add w.Range(w.Cells(5, 1), w.Cells(lastL, 15)).Formula Else oldLearners.Add Empty
        If lastP >= 5 Then oldPayments.Add pw.Range(pw.Cells(5, 1), pw.Cells(lastP, 14)).Formula Else oldPayments.Add Empty
    Next livre
    ITMJournalDebut: started = True
    ' Rattacher les paiements du dossier fusionne au matricule conserve dans tous les segments.
    For Each livre In livres
        Set pw = livre.Worksheets("PAIEMENTS")
        For r = 5 To ITMDerniereLigne(pw)
            If ITMCleTexte(pw.Cells(r, 4).Value2) = matriculeFusion Then
                pw.Cells(r, 4).Value2 = matriculeConserver
                pw.Cells(r, 5).Value2 = CStr(keepW.Cells(keepRow, 2).Value2) & " " & CStr(keepW.Cells(keepRow, 3).Value2) & " " & CStr(keepW.Cells(keepRow, 4).Value2)
                pw.Cells(r, 6).Value2 = CStr(keepW.Cells(keepRow, 8).Value2)
                nbPaiements = nbPaiements + 1
            End If
        Next r
    Next livre
    ' Retirer uniquement la fiche en doublon; le dossier conserve et son matricule restent inchanges.
    lastL = ITMDerniereLigne(mergeW)
    If mergeRow < lastL Then mergeW.Range(mergeW.Cells(mergeRow, 1), mergeW.Cells(lastL - 1, 15)).Formula = mergeW.Range(mergeW.Cells(mergeRow + 1, 1), mergeW.Cells(lastL, 15)).Formula
    mergeW.Range(mergeW.Cells(lastL, 1), mergeW.Cells(lastL, 15)).ClearContents
    mergeW.ListObjects(1).Resize mergeW.Range(mergeW.Cells(4, 1), mergeW.Cells(Application.Max(5, lastL - 1), 15))
    If keepW Is mergeW And mergeRow < keepRow Then keepRow = keepRow - 1
    apres = ITMJsonLigne(keepW, keepRow, 15)
    ITMJournalEcrire "FUSION_APPRENANTS", matriculeConserver, matriculeFusion, avant, apres, motif & " | " & CStr(nbPaiements) & " paiement(s) rattache(s) au matricule conserve."
    ' Save every touched segment. If a save fails, restore snapshots and attempt to save the rollback.
    For j = 1 To books.Count
        books(j).Save
        savedCount = j
    Next j
    ITMJournalConfirmer: started = False
    ITMAppFusionnerApprenants = "{""ok"":true,""matriculeConserve"":" & AJ(matriculeConserver) & ",""matriculeFusionne"":" & AJ(matriculeFusion) & ",""paiementsRattaches"":" & CStr(nbPaiements) & "}"
    GoTo Finally
EH:
    erreur = Err.Description
    On Error Resume Next
    If Not books Is Nothing Then
        For j = 1 To books.Count
            Set livre = books(j): Set w = livre.Worksheets("APPRENANTS"): Set pw = livre.Worksheets("PAIEMENTS")
            lastL = CLng(learnerLasts(j)): lastP = CLng(paymentLasts(j))
            If lastL >= 5 Then w.Range(w.Cells(5, 1), w.Cells(lastL, 15)).Formula = oldLearners(j)
            If ITMDerniereLigne(w) > lastL Then w.Range(w.Cells(lastL + 1, 1), w.Cells(ITMDerniereLigne(w), 15)).ClearContents
            w.ListObjects(1).Resize w.Range(w.Cells(4, 1), w.Cells(Application.Max(5, lastL), 15))
            If lastP >= 5 Then pw.Range(pw.Cells(5, 1), pw.Cells(lastP, 14)).Formula = oldPayments(j)
            If ITMDerniereLigne(pw) > lastP Then pw.Range(pw.Cells(lastP + 1, 1), pw.Cells(ITMDerniereLigne(pw), 14)).ClearContents
            pw.ListObjects(1).Resize pw.Range(pw.Cells(4, 1), pw.Cells(Application.Max(5, lastP), 14))
            livre.Save
        Next j
    End If
    If started Then ITMJournalAnnuler
    ITMAppFusionnerApprenants = "{""ok"":false,""error"":" & AJ(erreur) & "}"
Finally:
    On Error Resume Next
    If stateCaptured Then
        Application.EnableEvents = oldEvents
        Application.ScreenUpdating = oldScreen
        Application.Calculation = oldCalc
    End If
    On Error GoTo 0
End Function

Private Function ANumero(ByVal w As Worksheet) As String
    Dim n As Long, j As Long, alphabet As String, code As String
    Dim livres As Collection, livre As Workbook, disponible As Boolean
    Set livres = ITMClasseurs()
    alphabet = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ": Randomize
    For n = 1 To 1000
        code = ""
        For j = 1 To 5: code = code & Mid$(alphabet, 1 + Int(Rnd() * 36), 1): Next j
        disponible = True
        For Each livre In livres
            If Application.WorksheetFunction.CountIf(livre.Worksheets("PAIEMENTS").Columns(3), code) > 0 Then disponible = False
        Next livre
        If disponible Then ANumero = code: Exit Function
    Next n
    Err.Raise vbObjectError + 1014, , "Impossible de générer le numéro d'enregistrement."
End Function

Public Function ITMAppPaiement(ByVal matricule As String, ByVal montant As Double, Optional ByVal dateISO As String = "") As String
    Dim d As Object, p As Variant, f As Variant, w As Worksheet, r As Long, i As Long
    Dim numero As String, mensuel As Currency, annuel As Currency, paye As Currency, restant As Currency
    Dim disponible As Currency, verse As Currency, du As Currency, cumule As Currency, nb As Long, id As Long
    Dim m As Variant, cle As String, actif As Boolean, erreur As String, dateRecu As Date
    Dim livres As Collection, livre As Workbook
    On Error GoTo EH
    dateRecu = APaymentDate(dateISO)
    matricule = ITMCleTexte(matricule)
    Set d = ITMChargerDonnees()
    If Not d("personnes").Exists(matricule) Then Err.Raise vbObjectError + 1015, , "Apprenant invalide ou en double : vérifiez son dossier."
    p = d("personnes")(matricule): f = d("promotions")(p(3)): mensuel = CCur(f(0)): annuel = ITMPECAnnuelPayant(matricule, CStr(p(6)), mensuel, CCur(f(1)))
    If montant <= 0 Or montant <> Fix(montant) Then Err.Raise vbObjectError + 1016, , "Le montant doit être un nombre entier de FC supérieur à zéro."
    Set livres = ITMClasseurs()
    For Each livre In livres
    Set w = livre.Worksheets("PAIEMENTS")
    For i = 5 To ITMDerniereLigne(w)
        If ITMCleTexte(w.Cells(i, 4).Value2) = matricule Then
            If Not d("mois").Exists(ITMCleTexte(w.Cells(i, 7).Value2)) Or Not IsNumeric(w.Cells(i, 9).Value2) Or CStr(w.Cells(i, 9).Value2) = "" Then Err.Raise vbObjectError + 1017, , "Un paiement incomplet existe pour cet apprenant. Rapprochez le reçu avant un nouveau versement."
            If CCur(w.Cells(i, 9).Value2) < 0 Then Err.Raise vbObjectError + 1018, , "Un montant négatif doit être vérifié pour cet apprenant."
        End If
        If IsNumeric(w.Cells(i, 1).Value2) Then If CLng(w.Cells(i, 1).Value2) > id Then id = CLng(w.Cells(i, 1).Value2)
    Next i
    Next livre
    Set w = ThisWorkbook.Worksheets("PAIEMENTS")
    For Each m In d("mois").Keys
        cle = matricule & "|" & CStr(m)
        If d("sommes").Exists(cle) Then paye = paye + CCur(d("sommes")(cle))
    Next m
    If montant > annuel - paye Then Err.Raise vbObjectError + 1019, , "Le montant dépasse le reste annuel (" & Format$(annuel - paye, "#,##0") & " FC)."
    numero = ANumero(w): disponible = CCur(montant): cumule = paye
    ITMDebutSaisie w: actif = True: ITMJournalDebut
    For Each m In ITMPECMois(matricule, CStr(p(6)))
        If disponible <= 0 Then Exit For
        If Not d("mois").Exists(ITMCleTexte(m)) Then Err.Raise vbObjectError + 1020, , "Mois scolaire manquant dans PARAMETRES."
        cle = matricule & "|" & ITMCleTexte(m): du = mensuel - ITMPECMontant(matricule, CStr(p(6)), CStr(m), mensuel)
        If d("sommes").Exists(cle) Then du = du - CCur(d("sommes")(cle))
        If du > 0 Then
            verse = disponible: If verse > du Then verse = du
            restant = du - verse: cumule = cumule + verse
            r = ITMReserverLigne(w): id = id + 1
            w.Cells(r, 1).Value2 = id: w.Cells(r, 2).Value = dateRecu: w.Cells(r, 2).NumberFormat = "dd/mm/yyyy"
            w.Cells(r, 3).NumberFormat = "@": w.Cells(r, 3).Value2 = numero
            w.Cells(r, 4).NumberFormat = "@": w.Cells(r, 4).Value2 = matricule
            w.Cells(r, 5).NumberFormat = "@": w.Cells(r, 5).Value2 = p(0) & " " & p(1) & " " & p(2)
            w.Cells(r, 6).NumberFormat = "@": w.Cells(r, 6).Value2 = p(3)
            w.Cells(r, 7).Value2 = CStr(m): w.Cells(r, 8).Value2 = mensuel: w.Cells(r, 9).Value2 = verse
            w.Cells(r, 10).Value2 = restant: w.Cells(r, 11).Value2 = IIf(restant = 0, "PAYE", "LITIGE")
            w.Cells(r, 12).Value2 = cumule: w.Cells(r, 13).Value2 = annuel: w.Cells(r, 14).Value2 = Application.Max(0, annuel - cumule)
            w.Range(w.Cells(r, 8), w.Cells(r, 14)).NumberFormat = "#,##0"
            disponible = disponible - verse: nb = nb + 1
        End If
    Next m
    If disponible <> 0 Or nb = 0 Then Err.Raise vbObjectError + 1021, , "Le versement ne peut pas être entièrement réparti : enregistrement annulé."
    ITMJournalEcrire "PAIEMENT", matricule, numero, "", CStr(montant), "Recu valide a la caisse du " & Format$(dateRecu, "dd/mm/yyyy")
    ITMValiderSaisie: actif = False: ITMJournalConfirmer
    ITMAppPaiement = "{""ok"":true,""numero"":" & AJ(numero) & ",""montant"":" & AJ(CCur(montant)) & ",""mois"":" & AJ(nb) & "}"
    Exit Function
EH:
    erreur = Err.Description
    Err.Clear
    On Error Resume Next
    If actif Then
        ITMCollecterErreurAnnulation erreur
        ITMJournalAnnuler
        ITMCollecterErreurAnnulation erreur
        ITMAnnulerSaisie
        ITMCollecterErreurAnnulation erreur
    End If
    On Error GoTo 0
    ITMAppPaiement = "{""ok"":false,""error"":" & AJ(erreur) & "}"
End Function

Private Function APaymentDate(ByVal valeur As String) As Date
    Dim d As Date
    If valeur = "" Then
        APaymentDate = Date
        Exit Function
    End If
    If Not valeur Like "####-##-##" Then Err.Raise vbObjectError + 1090, , "Date du reçu invalide."
    d = DateSerial(CInt(Left$(valeur, 4)), CInt(Mid$(valeur, 6, 2)), CInt(Right$(valeur, 2)))
    If Format$(d, "yyyy-mm-dd") <> valeur Then Err.Raise vbObjectError + 1091, , "Date du reçu invalide."
    If d < DateSerial(1900, 1, 1) Or d > Date Then Err.Raise vbObjectError + 1092, , "La date du reçu doit être antérieure ou égale à aujourd'hui."
    APaymentDate = d
End Function

Public Function ITMAppScoliaImport(ByVal reference As String, ByVal identite As String, ByVal moisTexte As String, ByVal montantsTexte As String, ByVal dateISO As String, ByVal creer As Boolean) As String
    Dim livres As Collection, livre As Workbook, w As Worksheet, a As Worksheet, log As Worksheet
    Dim champs As Variant, mois As Variant, valeurs As Variant, p As Variant, f As Variant, d As Object
    Dim mat As String, numero As String, erreur As String, m As String, cle As String
    Dim i As Long, j As Long, r As Long, ar As Long, ancienDernier As Long, id As Long, nb As Long, actif As Boolean, ajoute As Boolean
    Dim montant As Currency, total As Currency, paye As Currency, annuel As Currency, cumule As Currency, dateRecu As Date, naissance As Date
    On Error GoTo EH
    If Len(reference) <> 32 Or reference Like "*[!a-f0-9]*" Then Err.Raise vbObjectError + 1300, , "Reference de transfert invalide."
    Set livres = ITMClasseurs()
    For Each livre In livres
        Set log = Nothing
        On Error Resume Next: Set log = livre.Worksheets("JOURNAL_ITM"): On Error GoTo EH
        If Not log Is Nothing Then
            For i = 2 To log.Cells(log.Rows.Count, 1).End(xlUp).Row
                If CStr(log.Cells(i, 3).Value2) = "SCOLIA_TRANSFERT" And CStr(log.Cells(i, 5).Value2) = reference Then
                    ITMAppScoliaImport = "{""ok"":true,""already"":true,""matricule"":" & AJ(log.Cells(i, 4).Value2) & "}": Exit Function
                End If
            Next i
        End If
    Next livre
    champs = Split(identite, vbTab): mois = Split(moisTexte, "|"): valeurs = Split(montantsTexte, "|")
    If UBound(champs) <> 12 Or UBound(mois) <> UBound(valeurs) Or UBound(mois) > 11 Then Err.Raise vbObjectError + 1301, , "Contenu du transfert incomplet."
    If Trim$(champs(1)) = "" Or Trim$(champs(2)) = "" Or Trim$(champs(3)) = "" Or Trim$(champs(4)) = "" Then Err.Raise vbObjectError + 1302, , "Nom, post-nom, prenom et promotion obligatoires."
    If Replace(champs(5), " ", "") <> Replace(GAnneeScolia(), " ", "") Then Err.Raise vbObjectError + 1303, , "Le transfert concerne une autre annee scolaire. Connectez sa base."
    dateRecu = APaymentDate(dateISO): Set d = ITMChargerDonnees()
    If Not d("promotions").Exists(ITMPromotion(champs(4))) Then Err.Raise vbObjectError + 1304, , "Promotion inconnue dans ce classeur. Appliquez la configuration SCOLIA."
    mat = ITMCleTexte(champs(0)): nb = 0
    For Each livre In livres
        Set a = livre.Worksheets("APPRENANTS")
        For i = 5 To ITMDerniereLigne(a)
            If AIdentiteCle(a.Cells(i, 2).Value2) = AIdentiteCle(champs(1)) And AIdentiteCle(a.Cells(i, 3).Value2) = AIdentiteCle(champs(2)) And AIdentiteCle(a.Cells(i, 4).Value2) = AIdentiteCle(champs(3)) And ITMPromotion(a.Cells(i, 8).Value2) = ITMPromotion(champs(4)) And Replace(CStr(a.Cells(i, 13).Value2), " ", "") = Replace(champs(5), " ", "") Then
                If mat <> "" And mat <> ITMCleTexte(a.Cells(i, 1).Value2) Then Err.Raise vbObjectError + 1305, , "Matricule et identite du transfert ne correspondent pas."
                mat = ITMCleTexte(a.Cells(i, 1).Value2): nb = nb + 1
            End If
        Next i
    Next livre
    If nb > 1 Then Err.Raise vbObjectError + 1306, , "Plusieurs apprenants correspondent : rapprochez les doublons."
    If nb = 0 And Not creer Then Err.Raise vbObjectError + 1307, , "Apprenant inconnu : transfert a conserver en brouillon."
    If nb = 0 Then
        If mat <> "" Then Err.Raise vbObjectError + 1308, , "Matricule inconnu : rapprochez le dossier avant import."
        mat = AMatricule(champs(4))
    End If
    f = d("promotions")(ITMPromotion(champs(4))): annuel = CCur(f(1))
    For j = 0 To UBound(mois)
        m = ITMCleTexte(mois(j)): cle = mat & "|" & m
        If Not d("mois").Exists(m) Or valeurs(j) Like "*[!0-9]*" Or valeurs(j) = "" Then Err.Raise vbObjectError + 1309, , "Mois ou montant invalide."
        For i = 0 To j - 1
            If ITMCleTexte(mois(i)) = m Then Err.Raise vbObjectError + 1310, , "Mois duplique dans le transfert."
        Next i
        montant = CCur(valeurs(j)): paye = 0
        If d("sommes").Exists(cle) Then paye = CCur(d("sommes")(cle))
        If montant <= 0 Or montant > CCur(f(0)) - paye - ITMPECMontant(mat, champs(5), m, CCur(f(0))) Then Err.Raise vbObjectError + 1311, , "Mensualite deja soldee ou montant superieur au reste."
        total = total + montant
    Next j
    paye = 0
    For Each p In d("sommes").Keys
        If Split(CStr(p), "|")(0) = mat Then paye = paye + CCur(d("sommes")(p))
    Next p
    If total > annuel - paye Then Err.Raise vbObjectError + 1312, , "Transfert superieur au reste annuel."
    Set w = ThisWorkbook.Worksheets("PAIEMENTS"): Set a = ThisWorkbook.Worksheets("APPRENANTS")
    For Each livre In livres
        For i = 5 To ITMDerniereLigne(livre.Worksheets("PAIEMENTS"))
            If IsNumeric(livre.Worksheets("PAIEMENTS").Cells(i, 1).Value2) Then If CLng(livre.Worksheets("PAIEMENTS").Cells(i, 1).Value2) > id Then id = CLng(livre.Worksheets("PAIEMENTS").Cells(i, 1).Value2)
        Next i
    Next livre
    numero = ANumero(w): ITMDebutSaisie w: actif = True: ITMJournalDebut
    If nb = 0 Then
        ancienDernier = ITMDerniereLigne(a): ar = Application.Max(5, ancienDernier + 1)
        If Application.CountA(a.Range(a.Cells(ar, 1), a.Cells(ar, 15))) <> 0 Then Err.Raise vbObjectError + 1313, , "Ligne apprenant non vide."
        a.ListObjects(1).Resize a.Range(a.Cells(4, 1), a.Cells(ar, 15)): ajoute = True
        a.Range(a.Cells(ar, 1), a.Cells(ar, 15)).NumberFormat = "@"
        a.Cells(ar, 1).Value2 = mat: a.Cells(ar, 2).Value2 = UCase$(champs(1)): a.Cells(ar, 3).Value2 = UCase$(champs(2)): a.Cells(ar, 4).Value2 = UCase$(champs(3))
        a.Cells(ar, 5).Value2 = champs(6)
        If champs(7) <> "" Then
            If Not ITMLireDate(champs(7), naissance) Then Err.Raise vbObjectError + 1331, , "Date de naissance incorrecte."
            a.Cells(ar, 6).Value = naissance: a.Cells(ar, 6).NumberFormat = "dd/mm/yyyy"
        End If
        a.Cells(ar, 7).Value2 = champs(8): a.Cells(ar, 8).Value2 = ITMPromotion(champs(4)): a.Cells(ar, 10).Value2 = champs(9): a.Cells(ar, 11).Value2 = champs(10)
        a.Cells(ar, 12).Value = Date: a.Cells(ar, 12).NumberFormat = "dd/mm/yyyy": a.Cells(ar, 13).Value2 = champs(5): a.Cells(ar, 14).Value2 = "ACTIF": a.Cells(ar, 15).Value2 = champs(11)
    End If
    cumule = paye
    For j = 0 To UBound(mois)
        montant = CCur(valeurs(j)): cumule = cumule + montant: id = id + 1: r = ITMReserverLigne(w)
        w.Cells(r, 1).Value2 = id: w.Cells(r, 2).Value = dateRecu: w.Cells(r, 2).NumberFormat = "dd/mm/yyyy"
        w.Cells(r, 3).NumberFormat = "@": w.Cells(r, 3).Value2 = numero: w.Cells(r, 4).NumberFormat = "@": w.Cells(r, 4).Value2 = mat
        w.Cells(r, 5).Value2 = champs(1) & " " & champs(2) & " " & champs(3): w.Cells(r, 6).Value2 = ITMPromotion(champs(4)): w.Cells(r, 7).Value2 = mois(j)
        w.Cells(r, 8).Value2 = CCur(f(0)): w.Cells(r, 9).Value2 = montant
        cle = mat & "|" & ITMCleTexte(mois(j)): paye = 0: If d("sommes").Exists(cle) Then paye = CCur(d("sommes")(cle))
        w.Cells(r, 10).Value2 = CCur(f(0)) - paye - montant - ITMPECMontant(mat, champs(5), CStr(mois(j)), CCur(f(0)))
        w.Cells(r, 11).Value2 = IIf(w.Cells(r, 10).Value2 = 0, "PAYE", "LITIGE"): w.Cells(r, 12).Value2 = cumule: w.Cells(r, 13).Value2 = annuel: w.Cells(r, 14).Value2 = annuel - cumule
    Next j
    ITMJournalEcrire "SCOLIA_TRANSFERT", mat, reference, numero, CStr(total), "Caisse : " & champs(12) & ". Paiement comptable importe une seule fois."
    ITMValiderSaisie: actif = False: ITMJournalConfirmer
    ITMAppScoliaImport = "{""ok"":true,""matricule"":" & AJ(mat) & ",""numero"":" & AJ(numero) & "}": Exit Function
EH:
    erreur = Err.Description: Err.Clear: On Error Resume Next
    If actif Then
        If ajoute Then a.Range(a.Cells(ar, 1), a.Cells(ar, 15)).ClearContents: a.ListObjects(1).Resize a.Range(a.Cells(4, 1), a.Cells(Application.Max(5, ancienDernier), 15))
        ITMJournalAnnuler: ITMCollecterErreurAnnulation erreur: ITMAnnulerSaisie: ITMCollecterErreurAnnulation erreur
    End If
    On Error GoTo 0
    ITMAppScoliaImport = "{""ok"":false,""error"":" & AJ(erreur) & "}"
End Function

Public Function ITMAppScoliaConfiguration(ByVal schoolId As String, ByVal annee As String, ByVal moisTexte As String, ByVal codesTexte As String, ByVal fraisTexte As String, ByVal confirmer As Boolean) As String
    Dim livre As Workbook, livres As Collection, w As Worksheet, p As Worksheet, mois As Variant, codes As Variant, frais As Variant, i As Long, j As Long, r As Long, trouve As Boolean
    Dim erreur As String
    On Error GoTo EH
    If Not confirmer Then Err.Raise vbObjectError + 1320, , "Confirmation necessaire."
    Set livres = ITMClasseurs()
    For Each livre In livres
        If ITMDerniereLigne(livre.Worksheets("PAIEMENTS")) >= 5 Then
            If Application.CountA(livre.Worksheets("PAIEMENTS").Range("A5:N" & ITMDerniereLigne(livre.Worksheets("PAIEMENTS")))) > 0 Then Err.Raise vbObjectError + 1321, , "Des paiements existent : calendrier et tarifs proteges. Utilisez une nouvelle base annuelle."
        End If
    Next livre
    mois = Split(moisTexte, "|"): codes = Split(codesTexte, "|"): frais = Split(fraisTexte, "|")
    If UBound(mois) > 11 Or UBound(codes) <> UBound(frais) Then Err.Raise vbObjectError + 1322, , "Configuration incomplete."
    For i = 0 To UBound(frais)
        If frais(i) Like "*[!0-9]*" Or frais(i) = "" Then Err.Raise vbObjectError + 1323, , "Tarif invalide."
        If CCur(frais(i)) <= 0 Then Err.Raise vbObjectError + 1324, , "Tarif positif obligatoire."
    Next i
    For Each livre In livres
        Set p = livre.Worksheets("APPRENANTS")
        For i = 5 To ITMDerniereLigne(p)
            If CStr(p.Cells(i, 1).Value2) <> "" Then
                If Replace(CStr(p.Cells(i, 13).Value2), " ", "") <> Replace(annee, " ", "") Then Err.Raise vbObjectError + 1332, , "Des apprenants d'une autre annee existent : utilisez la cloture annuelle."
                trouve = False
                For j = 0 To UBound(codes)
                    If ITMPromotion(p.Cells(i, 8).Value2) = ITMPromotion(codes(j)) Then trouve = True
                Next j
                If Not trouve Then Err.Raise vbObjectError + 1333, , "Une classe contient des apprenants : conservez-la dans la configuration."
            End If
        Next i
    Next livre
    For Each livre In livres
        Set w = livre.Worksheets("PARAMETRES"): w.Range("A5:A" & Application.Max(5, w.Cells(w.Rows.Count, 1).End(xlUp).Row)).ClearContents
        For i = 0 To UBound(mois): w.Cells(5 + i, 1).Value2 = mois(i): Next i
        Set p = livre.Worksheets("PROMOTIONS")
        p.Range("A5:H" & Application.Max(5, p.Cells(p.Rows.Count, 1).End(xlUp).Row)).ClearContents
        For i = 0 To UBound(codes)
            r = 5 + i
            p.Cells(r, 1).Value2 = ITMPromotion(codes(i)): p.Cells(r, 2).Value2 = codes(i): p.Cells(r, 5).Value2 = annee
            p.Cells(r, 6).Value2 = CCur(frais(i)): p.Cells(r, 7).Value2 = CCur(frais(i)) * (UBound(mois) + 1): p.Cells(r, 8).Value2 = "ACTIVE"
        Next i
        If p.ListObjects.Count = 1 Then p.ListObjects(1).Resize p.Range(p.Cells(4, 1), p.Cells(Application.Max(5, p.Cells(p.Rows.Count, 1).End(xlUp).Row), 8))
        On Error Resume Next
        livre.CustomDocumentProperties("ScoliaInstitutionId").Delete
        On Error GoTo EH
        livre.CustomDocumentProperties.Add "ScoliaInstitutionId", False, 4, schoolId
        livre.Save
    Next livre
    ITMAppScoliaConfiguration = "{""ok"":true}": Exit Function
EH:
    erreur = Err.Description: ITMAppScoliaConfiguration = "{""ok"":false,""error"":" & AJ(erreur) & "}"
End Function

Public Function ITMAppScoliaReferences() As String
    Dim livres As Collection, livre As Workbook, w As Worksheet, i As Long, resultat As String, first As Boolean
    first = True: resultat = "[": Set livres = ITMClasseurs()
    For Each livre In livres
        Set w = Nothing
        On Error Resume Next: Set w = livre.Worksheets("JOURNAL_ITM"): On Error GoTo 0
        If Not w Is Nothing Then
            For i = 2 To w.Cells(w.Rows.Count, 1).End(xlUp).Row
                If CStr(w.Cells(i, 3).Value2) = "SCOLIA_TRANSFERT" Then
                    If Not first Then resultat = resultat & ","
                    first = False: resultat = resultat & AJ(w.Cells(i, 5).Value2)
                End If
            Next i
        End If
    Next livre
    ITMAppScoliaReferences = resultat & "]"
End Function

Public Function ITMAppScoliaRecuProtege(ByVal numero As String) As String
    Dim livre As Workbook, livres As Collection, w As Worksheet, i As Long, erreur As String
    On Error GoTo EH
    Set livres = ITMClasseurs()
    For Each livre In livres
        Set w = Nothing
        On Error Resume Next
        Set w = livre.Worksheets("JOURNAL_ITM")
        On Error GoTo EH
        If Not w Is Nothing Then
            For i = 2 To w.Cells(w.Rows.Count, 1).End(xlUp).Row
                If CStr(w.Cells(i, 3).Value2) = "SCOLIA_TRANSFERT" And CStr(w.Cells(i, 6).Value2) = numero Then
                    ITMAppScoliaRecuProtege = "{""ok"":true,""protected"":true}": Exit Function
                End If
            Next i
        End If
    Next livre
    ITMAppScoliaRecuProtege = "{""ok"":true,""protected"":false}": Exit Function
EH:
    erreur = Err.Description: ITMAppScoliaRecuProtege = "{""ok"":false,""error"":" & AJ(erreur) & "}"
End Function
